/*! \file shk_cloud.c  (WIND-TUNNEL VERSION - replaces the planar-shock setup)
 *  \brief Cloud-wind problem matched to the FLASH CloudCrush configuration:
 *   a uniform Mach-M wind (problem/Mach, in units of the ambient sound speed)
 *   flows in +x1 past a static spherical cloud of overdensity drat in exact
 *   pressure equilibrium. Cloud radius fixed at 1.0, centered at the origin.
 *   Density and velocity are smoothed across the cloud edge with the same
 *   tanh profile FLASH uses (width = 0.1*rad).
 *   Suggested domain: x1 in [-3,17], x2 in [-5,5] (x3 in [-5,5] for 3D),
 *   giving 20x10(x10) cloud radii with the cloud 3 radii from the inflow,
 *   mirroring the FLASH CloudCrush box.
 *    - problem/Mach = wind Mach number (v_wind / c_s,ambient)
 *    - problem/drat = cloud/ambient density ratio (chi)
 *   Cloud material is labeled with the passive scalar s[0] when available.
 * PRIVATE FUNCTION PROTOTYPES:
 * shk_cloud_iib() - fixes BCs on L-x1 (left edge) of grid to the wind state */
/*============================================================================*/

#include <math.h>
#include <stdio.h>
#include "defs.h"
#include "athena.h"

#include "prototypes.h"

/* wind state shared with the IIB function */

static Real dl,pl,ul;
#ifdef MHD
static Real bxl,byl,bzl;
#endif /* MHD */

void shk_cloud_iib(GridS *pGrid);

/*=========================== PUBLIC FUNCTIONS ===============================*/
/*----------------------------------------------------------------------------*/
/* problem:  */

void problem(DomainS *pDomain)
{
  GridS *pGrid = pDomain->Grid;
  int i=0,j=0,k=0;
  int is,ie,js,je,ks,ke;
  Real x1,x2,x3,diag,rad,edge;
  Real Mach,drat,f,d,vx;
  Real dr,pr;

/* Read input parameters */

  rad  = 1.0;
  edge = 0.1*rad;              /* tanh smoothing width, matches FLASH */
  Mach = par_getd("problem","Mach");
  drat = par_getd("problem","drat");

/* Ambient (wind) state: rho=1, p=1 so c_s=sqrt(Gamma)~1.291 as in FLASH */

  dr = 1.0;
  pr = 1.0;

/* Wind state shared with the boundary function */

  dl = dr;
  pl = pr;
#ifdef ISOTHERMAL
  ul = Mach*Iso_csound;
#else
  ul = Mach*sqrt(Gamma*pr/dr);
#endif

/* Initialize the grid: wind everywhere, static smoothed cloud at origin */

  is = pGrid->is;  ie = pGrid->ie;
  js = pGrid->js;  je = pGrid->je;
  ks = pGrid->ks;  ke = pGrid->ke;

  for (k=ks; k<=ke; k++) {
    for (j=js; j<=je; j++) {
      for (i=is; i<=ie; i++) {
        cc_pos(pGrid,i,j,k,&x1,&x2,&x3);
        diag = sqrt(x1*x1 + x2*x2 + x3*x3);

        f  = 0.5*(1.0 - tanh((diag - rad)/edge));
        d  = dr*(1.0 + (drat - 1.0)*f);
        /* velocity transition sits OUTSIDE the dense edge (1.3*rad, as in
         * Grete & Gronke's AthenaPK cloud): dense+fast cells would suffer
         * catastrophic total-energy cancellation at high drat */
        vx = ul*(1.0 - 0.5*(1.0 - tanh((diag - 1.3*rad)/edge)));

        pGrid->U[k][j][i].d  = d;
        pGrid->U[k][j][i].M1 = d*vx;
        pGrid->U[k][j][i].M2 = 0.0;
        pGrid->U[k][j][i].M3 = 0.0;
#ifdef ADIABATIC
        pGrid->U[k][j][i].E = pr/Gamma_1 + 0.5*d*vx*vx;
#endif
#if (NSCALARS > 0)
        pGrid->U[k][j][i].s[0] = drat*f;
#endif
      }
    }
  }

/* Set IIB value function pointer */

  if (pDomain->Disp[0] == 0) bvals_mhd_fun(pDomain,left_x1,shk_cloud_iib);

  return;
}

/*! \fn void shk_cloud_iib(GridS *pGrid)
 *  \brief Sets boundary condition on left X boundary (iib):
 *  quantities held fixed at the incoming wind state */

void shk_cloud_iib(GridS *pGrid)
{
  int i=0,j=0,k=0;
  int js,je,ks,ke;

  js = pGrid->js; je = pGrid->je;
  ks = pGrid->ks; ke = pGrid->ke;

  for (k=ks; k<=ke; k++) {
    for (j=js; j<=je; j++) {
      for (i=1; i<=nghost; i++) {
        pGrid->U[k][j][i].d  = dl;
        pGrid->U[k][j][i].M1 = ul*dl;
        pGrid->U[k][j][i].M2 = 0.0;
        pGrid->U[k][j][i].M3 = 0.0;
#ifdef ADIABATIC
        pGrid->U[k][j][i].E = pl/Gamma_1 + 0.5*dl*(ul*ul);
#endif
#if (NSCALARS > 0)
        pGrid->U[k][j][i].s[0] = 0.0;
#endif
      }
    }
  }
}
