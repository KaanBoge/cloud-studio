!!****if* source/Simulation/SimulationMain/CloudStatic/Simulation_initBlock
!!
!! NAME
!!  Simulation_initBlock
!!
!! SYNOPSIS
!!  call Simulation_initBlock(integer(IN) :: blockId)
!!
!! DESCRIPTION
!!  Initializes a cold, dense cloud of radius sim_rCloud centered at
!!  (sim_xctr, sim_yctr, sim_zctr) inside a hot, dilute ambient medium.
!!  The pressure is uniform (pressure equilibrium) and all velocities are
!!  zero, so the configuration should remain static.  The cloud edge is
!!  smoothed with a tanh profile of width sim_smoothWidth*sim_rCloud to
!!  avoid a perfectly sharp contact discontinuity.
!!
!!  Density profile:
!!    rho(r) = rho_amb + (rho_cl - rho_amb) * 0.5*(1 - tanh((r - r_cl)/delta))
!!
!!  Since P is uniform and T ~ P/rho, the cloud is colder than the
!!  ambient medium by the density contrast chi = rho_cl/rho_amb.
!!
!! ARGUMENTS
!!  blockId -  The number of the block to initialize
!!
!!***

subroutine Simulation_initBlock(blockId)

  use Simulation_data, ONLY: sim_pAmbient, sim_rhoAmbient, sim_rhoCloud, &
       sim_rCloud, sim_smoothWidth, sim_xCenter, sim_yCenter, sim_zCenter, &
       sim_gamma, sim_smallP, sim_smallT, sim_gCell
  use Grid_interface, ONLY : Grid_getBlkIndexLimits, Grid_getCellCoords, &
       Grid_putPointData

  implicit none

#include "constants.h"
#include "Flash.h"

  integer, intent(IN) :: blockId

  integer :: i, j, k, sizeX, sizeY, sizeZ, istat
  integer, dimension(2,MDIM) :: blkLimits, blkLimitsGC
  integer, dimension(MDIM)   :: axis
  real, allocatable, dimension(:) :: xCoord, yCoord, zCoord
  real :: xDist, yDist, zDist, radius, delta
  real :: rho, p, eint, ener

  call Grid_getBlkIndexLimits(blockId, blkLimits, blkLimitsGC)

  sizeX = blkLimitsGC(HIGH,IAXIS)
  sizeY = blkLimitsGC(HIGH,JAXIS)
  sizeZ = blkLimitsGC(HIGH,KAXIS)

  allocate(xCoord(sizeX), stat=istat)
  allocate(yCoord(sizeY), stat=istat)
  allocate(zCoord(sizeZ), stat=istat)
  xCoord = 0.0
  yCoord = 0.0
  zCoord = 0.0

  if (NDIM == 3) call Grid_getCellCoords(KAXIS, blockId, CENTER, sim_gCell, zCoord, sizeZ)
  if (NDIM >= 2) call Grid_getCellCoords(JAXIS, blockId, CENTER, sim_gCell, yCoord, sizeY)
  call Grid_getCellCoords(IAXIS, blockId, CENTER, sim_gCell, xCoord, sizeX)

  ! width of the smoothed cloud edge; avoid division by zero for sharp edge
  delta = max(sim_smoothWidth * sim_rCloud, tiny(1.0))

  do k = blkLimitsGC(LOW,KAXIS), blkLimitsGC(HIGH,KAXIS)
     zDist = 0.0
     if (NDIM == 3) zDist = zCoord(k) - sim_zCenter
     do j = blkLimitsGC(LOW,JAXIS), blkLimitsGC(HIGH,JAXIS)
        yDist = 0.0
        if (NDIM >= 2) yDist = yCoord(j) - sim_yCenter
        do i = blkLimitsGC(LOW,IAXIS), blkLimitsGC(HIGH,IAXIS)
           xDist = xCoord(i) - sim_xCenter

           radius = sqrt(xDist**2 + yDist**2 + zDist**2)

           ! smoothed density profile: cloud interior -> sim_rhoCloud,
           ! far field -> sim_rhoAmbient
           rho = sim_rhoAmbient + (sim_rhoCloud - sim_rhoAmbient) * &
                 0.5 * (1.0 - tanh((radius - sim_rCloud)/delta))

           ! uniform pressure, zero velocity everywhere
           p    = max(sim_pAmbient, sim_smallP)
           eint = p / ((sim_gamma - 1.0) * rho)
           ener = eint      ! no kinetic energy: the setup is static

           axis(IAXIS) = i
           axis(JAXIS) = j
           axis(KAXIS) = k

           call Grid_putPointData(blockId, CENTER, DENS_VAR, EXTERIOR, axis, rho)
           call Grid_putPointData(blockId, CENTER, PRES_VAR, EXTERIOR, axis, p)
           call Grid_putPointData(blockId, CENTER, VELX_VAR, EXTERIOR, axis, 0.0)
           call Grid_putPointData(blockId, CENTER, VELY_VAR, EXTERIOR, axis, 0.0)
           call Grid_putPointData(blockId, CENTER, VELZ_VAR, EXTERIOR, axis, 0.0)
           call Grid_putPointData(blockId, CENTER, ENER_VAR, EXTERIOR, axis, ener)
#ifdef EINT_VAR
           call Grid_putPointData(blockId, CENTER, EINT_VAR, EXTERIOR, axis, eint)
#endif
#ifdef GAME_VAR
           call Grid_putPointData(blockId, CENTER, GAME_VAR, EXTERIOR, axis, sim_gamma)
#endif
#ifdef GAMC_VAR
           call Grid_putPointData(blockId, CENTER, GAMC_VAR, EXTERIOR, axis, sim_gamma)
#endif
#ifdef TEMP_VAR
           call Grid_putPointData(blockId, CENTER, TEMP_VAR, EXTERIOR, axis, sim_smallT)
#endif
        enddo
     enddo
  enddo

  deallocate(xCoord)
  deallocate(yCoord)
  deallocate(zCoord)

end subroutine Simulation_initBlock
