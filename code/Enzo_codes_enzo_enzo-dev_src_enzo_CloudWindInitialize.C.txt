/***********************************************************************
/
/  INITIALIZE A CLOUD-WIND (CLOUD CRUSHING) PROBLEM
/
/  written by: Claude (cloud-crushing code comparison, HARP/Horizon)
/  date:       August 2026
/
/  PURPOSE: Mach-2 wind hitting a static, smooth-edged overdense cloud.
/    Matched setup shared with FLASH CloudCrush and Athena cloud_wind:
/      ambient/wind: rho = 1, P = 1, vx = Mach*c_s flowing +x
/      cloud: overdensity chi, static, same pressure, radius R
/      smooth edge: f = 0.5*(1 - tanh((r - R)/w))
/        rho = 1 + (chi-1)*f ,  vx = v_wind*(1-f)
/    Left x boundary: inflow (fixed wind); all others: outflow.
/    Optional passive tracer of cloud material (Metallicity colour
/    field, "Metal_Density" = f * rho).
/
/  RETURNS: SUCCESS or FAIL
/
************************************************************************/

#include <string.h>
#include <stdio.h>
#include <math.h>
#include "ErrorExceptions.h"
#include "macros_and_parameters.h"
#include "typedefs.h"
#include "global_data.h"
#include "Fluxes.h"
#include "GridList.h"
#include "ExternalBoundary.h"
#include "Grid.h"
#include "Hierarchy.h"
#include "TopGridData.h"

int CloudWindInitialize(FILE *fptr, FILE *Outfptr, HierarchyEntry &TopGrid,
			TopGridData &MetaData, ExternalBoundary &Exterior)
{
  char *DensName  = "Density";
  char *TEName    = "TotalEnergy";
  char *GEName    = "GasEnergy";
  char *Vel1Name  = "x-velocity";
  char *Vel2Name  = "y-velocity";
  char *Vel3Name  = "z-velocity";
  char *MetalName = "Metal_Density";

  /* declarations */

  char line[MAX_LINE_LENGTH];
  int dim, ret;

  /* set default parameters (code units; c_s,ambient = sqrt(Gamma) ) */

  float CloudWindAmbientDensity  = 1.0;
  float CloudWindAmbientPressure = 1.0;
  float CloudWindVelocity        = 2.582;   // Mach 2.0 for gamma=5/3, P=rho=1
  float CloudWindChi             = 10.0;    // cloud overdensity
  FLOAT CloudWindCloudRadius     = 1.0;
  FLOAT CloudWindSmoothWidth     = 0.1;
  FLOAT CloudWindCenter[MAX_DIMENSION];
  int   CloudWindUseTracer       = TRUE;

  for (dim = 0; dim < MAX_DIMENSION; dim++)
    CloudWindCenter[dim] = 0.0;

  /* read input from file */

  while (fgets(line, MAX_LINE_LENGTH, fptr) != NULL) {

    ret = 0;

    ret += sscanf(line, "CloudWindAmbientDensity = %"FSYM,
		  &CloudWindAmbientDensity);
    ret += sscanf(line, "CloudWindAmbientPressure = %"FSYM,
		  &CloudWindAmbientPressure);
    ret += sscanf(line, "CloudWindVelocity = %"FSYM, &CloudWindVelocity);
    ret += sscanf(line, "CloudWindChi = %"FSYM, &CloudWindChi);
    ret += sscanf(line, "CloudWindCloudRadius = %"PSYM, &CloudWindCloudRadius);
    ret += sscanf(line, "CloudWindSmoothWidth = %"PSYM, &CloudWindSmoothWidth);
    ret += sscanf(line, "CloudWindCenter = %"PSYM" %"PSYM" %"PSYM,
		  CloudWindCenter, CloudWindCenter+1, CloudWindCenter+2);
    ret += sscanf(line, "CloudWindUseTracer = %"ISYM, &CloudWindUseTracer);

    /* if the line is suspicious, issue a warning */

    if (ret == 0 && strstr(line, "=") && strstr(line, "CloudWind"))
      fprintf(stderr, "CloudWind warning: the following parameter line was not interpreted:\n%s\n", line);

  }

  /* set up the top grid */

  if (TopGrid.GridData->
      CloudWindInitializeGrid(CloudWindAmbientDensity,
			      CloudWindAmbientPressure,
			      CloudWindVelocity,
			      CloudWindChi,
			      CloudWindCloudRadius,
			      CloudWindSmoothWidth,
			      CloudWindCenter,
			      CloudWindUseTracer) == FAIL) {
    ENZO_FAIL("Error in CloudWindInitializeGrid.");
  }

  /* set up field names and units (order must match the grid initializer) */

  int labelCounter = 0;

  int DensNum = labelCounter;
  DataUnits[labelCounter] = NULL;
  DataLabel[labelCounter++] = DensName;

  int Vel1Num = labelCounter;
  DataUnits[labelCounter] = NULL;
  DataLabel[labelCounter++] = Vel1Name;

  int Vel2Num = -1;
  int Vel3Num = -1;
  if (TopGrid.GridData->GetGridRank() > 1 || HydroMethod > 2) {
    Vel2Num = labelCounter;
    DataUnits[labelCounter] = NULL;
    DataLabel[labelCounter++] = Vel2Name;
    if (TopGrid.GridData->GetGridRank() > 2 || HydroMethod > 2) {
      Vel3Num = labelCounter;
      DataUnits[labelCounter] = NULL;
      DataLabel[labelCounter++] = Vel3Name;
    }
  }

  int TENum = labelCounter;
  DataUnits[labelCounter] = NULL;
  DataLabel[labelCounter++] = TEName;

  int IENum = -1;
  if (DualEnergyFormalism) {
    IENum = labelCounter;
    DataUnits[labelCounter] = NULL;
    DataLabel[labelCounter++] = GEName;
  }

  int MetalNum = -1;
  if (CloudWindUseTracer) {
    MetalNum = labelCounter;
    DataUnits[labelCounter] = NULL;
    DataLabel[labelCounter++] = MetalName;
  }

  /* Initialize the exterior: fixed wind inflow on inner x face,
     outflow everywhere else. */

  Exterior.Prepare(TopGrid.GridData);

  float InflowValue[labelCounter], Dummy[labelCounter];
  for (int j = 0; j < labelCounter; j++) {
    InflowValue[j] = 0.0;
    Dummy[j] = 0.0;
  }

  InflowValue[DensNum] = CloudWindAmbientDensity;
  InflowValue[Vel1Num] = CloudWindVelocity;
  InflowValue[TENum]   = CloudWindAmbientPressure /
    ((Gamma - 1.0)*CloudWindAmbientDensity);
  if (HydroMethod != 2)
    InflowValue[TENum] += 0.5*POW(CloudWindVelocity, 2);
  if (IENum > -1)
    InflowValue[IENum] = CloudWindAmbientPressure /
      ((Gamma - 1.0)*CloudWindAmbientDensity);
  if (MetalNum > -1)
    InflowValue[MetalNum] = tiny_number;

  if (Exterior.InitializeExternalBoundaryFace(0, inflow, outflow, InflowValue,
					      Dummy) == FAIL) {
    ENZO_FAIL("Error in InitializeExternalBoundaryFace.");
  }

  if (MetaData.TopGridRank > 1)
    Exterior.InitializeExternalBoundaryFace(1, outflow, outflow, Dummy, Dummy);
  if (MetaData.TopGridRank > 2)
    Exterior.InitializeExternalBoundaryFace(2, outflow, outflow, Dummy, Dummy);

  /* Write parameters to parameter output file */

  if (MyProcessorNumber == ROOT_PROCESSOR) {
    fprintf(Outfptr, "CloudWindAmbientDensity  = %"FSYM"\n",
	    CloudWindAmbientDensity);
    fprintf(Outfptr, "CloudWindAmbientPressure = %"FSYM"\n",
	    CloudWindAmbientPressure);
    fprintf(Outfptr, "CloudWindVelocity        = %"FSYM"\n", CloudWindVelocity);
    fprintf(Outfptr, "CloudWindChi             = %"FSYM"\n", CloudWindChi);
    fprintf(Outfptr, "CloudWindCloudRadius     = %"GOUTSYM"\n",
	    CloudWindCloudRadius);
    fprintf(Outfptr, "CloudWindSmoothWidth     = %"GOUTSYM"\n",
	    CloudWindSmoothWidth);
    fprintf(Outfptr, "CloudWindCenter          = %"GOUTSYM" %"GOUTSYM" %"GOUTSYM"\n",
	    CloudWindCenter[0], CloudWindCenter[1], CloudWindCenter[2]);
    fprintf(Outfptr, "CloudWindUseTracer       = %"ISYM"\n", CloudWindUseTracer);
  }

  return SUCCESS;
}
