!! NAME
!!  Simulation_data
!!
!! SYNOPSIS
!!  use Simulation_data
!!
!! DESCRIPTION
!!  Stores the runtime parameters of the CloudStatic problem:
!!  a cold dense cloud in a hot dilute medium in pressure equilibrium.
!!

module Simulation_data

  implicit none

  !! Runtime parameters
  real, save :: sim_pAmbient, sim_rhoAmbient, sim_rhoCloud
  real, save :: sim_rCloud, sim_smoothWidth
  real, save :: sim_xCenter, sim_yCenter, sim_zCenter
  real, save :: sim_gamma, sim_smallP, sim_smallX, sim_smallT

  logical, save :: sim_gCell = .true.

end module Simulation_data
