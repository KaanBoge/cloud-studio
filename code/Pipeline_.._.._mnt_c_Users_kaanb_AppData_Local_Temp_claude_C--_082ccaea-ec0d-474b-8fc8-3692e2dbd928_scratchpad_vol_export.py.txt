"""Extract nested density isosurfaces from a 3D snapshot into a compact JS file
for the studio's interactive Volumes viewer.

Usage: vol_export.py <vol_id> <kind> <chi> <tcc> <label> <snapshot_file>
  kind: apk | flash | athena
Output: C:\\...\\CloudCrushing\\studio\\data\\vol_<vol_id>.js
Coordinates: x = streamwise (wind +x), cloud at origin, units of cloud radii.
"""
import base64
import json
import sys

import numpy as np
import yt
import matplotlib
from skimage.measure import marching_cubes

yt.set_log_level(40)

VID, KIND, CHI, TCC, LABEL, FN = (
    sys.argv[1], sys.argv[2], int(sys.argv[3]), float(sys.argv[4]), sys.argv[5], sys.argv[6])
OUT = "/mnt/c/Users/kaanb/CloudCrushing/studio/data/vol_%s.js" % VID

CLIM = {10: (-1.0, 1.5), 100: (-1.0, 2.5), 1000: (-1.0, 3.5)}
lo, hi = CLIM.get(CHI, (-1.0, 1.5))

# isosurface levels: wind shock structure + outer cloud + dense core
LEVELS = [1.8, 0.15 * CHI, 0.6 * CHI]
OPAC = [0.18, 0.45, 0.95]

ds = yt.load(FN)
dims = [int(v) for v in ds.domain_dimensions]
lvl = 0
if KIND == "flash":
    lvl = min(4, ds.index.max_level)
    dims = [d * 2 ** lvl for d in dims]
cg = ds.covering_grid(lvl, ds.domain_left_edge, dims)
rho = np.array(cg[("gas", "density")], dtype=np.float32)
dle = np.array(ds.domain_left_edge, dtype=float)
dre = np.array(ds.domain_right_edge, dtype=float)

if KIND == "apk":
    # axes (x1 transverse, x2 streamwise, x3 z) -> (x2, x1, x3)
    rho = np.transpose(rho, (1, 0, 2))
    lo_xyz = [dle[1], dle[0], dle[2]]
    hi_xyz = [dre[1], dre[0], dre[2]]
else:
    lo_xyz = list(dle[:3])
    hi_xyz = list(dre[:3])
    if KIND == "flash":
        # FLASH box 2x1x1 with r_c=0.1, cloud at (0.3,0.5,0.5) -> cloud radii
        rho = rho  # handled by coordinate transform below
        lo_xyz = [(lo_xyz[0] - 0.3) / 0.1, (lo_xyz[1] - 0.5) / 0.1, (lo_xyz[2] - 0.5) / 0.1]
        hi_xyz = [(hi_xyz[0] - 0.3) / 0.1, (hi_xyz[1] - 0.5) / 0.1, (hi_xyz[2] - 0.5) / 0.1]

# cap the marching-cubes grid so surfaces stay web-sized
while max(rho.shape) > 288:
    rho = rho[::2, ::2, ::2]

sp = [(hi_xyz[i] - lo_xyz[i]) / rho.shape[i] for i in range(3)]

viridis = matplotlib.colormaps["viridis"]
surfaces = []
t = float(ds.current_time)
for lev, op in zip(LEVELS, OPAC):
    if lev >= float(rho.max()) or lev <= float(rho.min()):
        continue
    try:
        verts, faces, _, _ = marching_cubes(rho, level=lev, step_size=1)
    except Exception as e:
        print("MC fail", lev, repr(e))
        continue
    # decimate huge surfaces by dropping every other triangle
    while len(faces) > 90000:
        faces = faces[::2]
    tri = verts[faces].reshape(-1, 3).astype(np.float32)
    for i in range(3):
        tri[:, i] = lo_xyz[i] + tri[:, i] * sp[i]
    frac = (np.log10(lev) - lo) / (hi - lo)
    r, g, b, _ = viridis(min(max(frac, 0.0), 1.0))
    surfaces.append({
        "level": lev, "opacity": op,
        "color": [round(r, 3), round(g, 3), round(b, 3)],
        "tris": len(faces),
        "pos": base64.b64encode(tri.tobytes()).decode(),
    })
    print("level %.2f -> %d triangles" % (lev, len(faces)))

meta = {"code": LABEL, "chi": CHI, "res": "x".join(str(d) for d in dims),
        "t": "%.1f t_cc" % (t / TCC)}
with open(OUT, "w") as f:
    f.write("VOL_DATA[%s]={meta:%s,ext:%s,surfaces:%s};\n" % (
        json.dumps(VID), json.dumps(meta),
        json.dumps([lo_xyz, hi_xyz]), json.dumps(surfaces)))
print("wrote", OUT)
